import Article, { articleMeta, conversationFields } from "../app/articles/content/featured/how-to-get-the-first-customers-for-my-startup-in-2026";

test("the page and real download preserve every completed field, supplied answer and event", () => {
  const html = render();
  const text = readFileSync("public/downloads/first-customer-conversation.txt", "utf8");
  expect(text).toBe(firstCustomerWorksheetText());
  expect(CUSTOMER_COMPLETED_FIELDS).toHaveLength(12);
  for (const field of CUSTOMER_COMPLETED_FIELDS) {
    expect(text).toContain(field.label + ":"); expect(text).toContain(field.value);
    expect(html).toContain(field.label); expect(html).toContain(field.value);
  }
  for (const note of CUSTOMER_NOTES) {
    expect(html).toContain(`data-customer-note="${note.id}"`);
    for (const key of ["question", "answer", "interpretation", "unknown"] as const) {
      expect(text).toContain(note[key]); expect(html).toContain(note[key]);
    }
  }
  for (const event of CUSTOMER_EVENTS) {
    expect(html).toContain(`data-customer-event="${event.id}"`);
    for (const key of ["date", "action", "source", "status"] as const) {
      expect(text).toContain(event[key]); expect(html).toContain(event[key]);
    }
  }
  for (const value of [CUSTOMER_PROVENANCE, CUSTOMER_REVIEW_STATUS, CUSTOMER_INVITATION, CUSTOMER_OUTLINE, CUSTOMER_DECLINE, CUSTOMER_DECISION]) {
    expect(text).toContain(value); expect(html).toContain(value);
  }
  expect(text).toContain("ACTUAL-CASE FOLLOW-UP — DO NOT PREFILL A SUCCESS");
  expect(html).toContain('download="first-customer-conversation.txt"');
});

test("the dated refusal supersedes the single-outline permission without inventing a price test", () => {
  expect(CUSTOMER_EVENTS.map(e => e.id)).toEqual(["E1", "E2", "E3", "E4", "E5"]);
  expect(CUSTOMER_EVENTS.map(e => e.date)).toEqual(["2026-09-14", "2026-09-16", "2026-09-17", "2026-09-18", "2026-09-21"]);
  expect(CUSTOMER_NOTES[5].answer).toContain("one outline on Thursday");
  expect(new Date(CUSTOMER_EVENTS[2].date + "T12:00:00Z").getUTCDay()).toBe(4);
  expect(CUSTOMER_EVENTS[2].source).toBe(CUSTOMER_OUTLINE);
  expect(CUSTOMER_EVENTS[3].source).toBe(CUSTOMER_DECLINE);
  expect(CUSTOMER_EVENTS[4].status).toBe(CUSTOMER_DECISION);
  expect(CUSTOMER_DECISION).toContain("cannot justify reminders or an owner approach");
  expect(CUSTOMER_NOTES[3].answer).toContain("before adding another tool");
  expect(CUSTOMER_NOTES[4].answer).toContain("not arranging an introduction");
  expect(CUSTOMER_COMPLETED_FIELDS[8].value).toContain("Do not infer a price objection: no price was shown");
  const html = render();
  expect(html).not.toContain("the draft looks useful");
  expect(html).not.toContain("The next step in this example is an agreed demonstration");
  expect(html).toContain("not one of the companion GTM guide’s 28 records");
});

test("stages and the four-week plan preserve the non-sale and unperformed later work", () => {
  const html = render(), text = firstCustomerWorksheetText();
  expect(CUSTOMER_STAGE_ROWS).toHaveLength(6);
  for (const row of CUSTOMER_STAGE_ROWS) for (const value of Object.values(row)) { expect(html).toContain(value); expect(text).toContain(value); }
  expect(CUSTOMER_STAGE_ROWS[2].state).toBe("No booking and no attendance.");
  expect(CUSTOMER_STAGE_ROWS[3].state).toBe("No trial, usage or completed pilot review.");
  expect(CUSTOMER_STAGE_ROWS[4].state).toBe("No priced offer shown; no accepted paid terms.");
  expect(CUSTOMER_STAGE_ROWS[5].state).toBe("No payment or completed delivery.");
  expect(CUSTOMER_WEEKS).toHaveLength(4);
  for (const week of CUSTOMER_WEEKS) for (const value of Object.values(week)) { expect(html).toContain(value); expect(text).toContain(value); }
  expect(CUSTOMER_WEEKS[2].decision).toContain("Not run in this example");
  expect(CUSTOMER_WEEKS[3].decision).toContain("Planned, not run here");
  expect(html).toContain("One participant is not a market or channel-conversion sample");
});

test("event preference is allowlisted, requires application and never includes interview notes", () => {
  for (const value of ["online", "melbourne"]) {
    const html = render("/articles/" + slug + "?event_format=" + value + "&prospect=PRIVATE_SENTINEL");
    expect(html).toContain(`href="/events?event_format=${value}"`);
    expect(html).not.toContain("PRIVATE_SENTINEL");
  }
  for (const value of ["all", "unexpected"]) expect(render("/articles/" + slug + "?event_format=" + value)).toContain('href="/events"');
  const html = render();
  expect(html).toContain(`action="/articles/${slug}" method="get"`);
  expect(html).toContain("Apply preference");
  expect(html).toContain("Your interview notes and worksheet are not uploaded or attached");
  expect(html).toContain("Opening a workspace requires an account");
});
