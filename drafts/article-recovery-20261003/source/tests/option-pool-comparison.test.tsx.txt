const example = { preMoney: 8, newMoney: 2, poolPercent: 10 };
const good = (input = example) => {
  const result = compareOptionPoolDilution(input);
  if (!result.ok) throw new Error(result.error);
  return result;
};

test('the fictional example distinguishes no pool, pre-money inclusion and later pro-rata dilution', () => {
  const result = good();
  expect(result.noPool).toEqual({ existingPercent: 80, investorPercent: 20, poolPercent: 0 });
  expect(result.preMoneyPool.existingPercent).toBeCloseTo(70, 10);
  expect(result.preMoneyPool.investorPercent).toBe(20);
  expect(result.postMoneyPool.existingPercent).toBeCloseTo(72, 10);
  expect(result.postMoneyPool.investorPercent).toBeCloseTo(18, 10);
  expect(result.preMoneyPool.poolPercent).toBe(10);
  expect(result.postMoneyPool.poolPercent).toBe(10);
  expect(result.existingDifferencePoints).toBeCloseTo(2, 10);
  expect(result.postMoneyPool.existingPercent / result.preMoneyPool.existingPercent - 1).not.toBeCloseTo(0.02, 4);
});

test('percentage equations agree with separately constructed share-count scenarios', () => {
  for (const preMoney of [1, 5, 8, 27]) for (const newMoney of [1, 2, 3, 10]) for (const poolPercent of [0, 1, 5, 10, 20]) {
    const p = poolPercent / 100;
    if (p >= preMoney / (preMoney + newMoney)) continue;
    const result = good({ preMoney, newMoney, poolPercent });
    const shares = 1_000_000;
    const ratio = newMoney / preMoney;
    const prePool = p * (1 + ratio) * shares / (1 - p * (1 + ratio));
    const prePrice = preMoney / (shares + prePool);
    const preNew = newMoney / prePrice;
    const preTotal = shares + prePool + preNew;
    expect(result.preMoneyPool.existingPercent).toBeCloseTo(shares / preTotal * 100, 9);
    expect(result.preMoneyPool.investorPercent).toBeCloseTo(preNew / preTotal * 100, 9);
    expect(result.preMoneyPool.poolPercent).toBeCloseTo(prePool / preTotal * 100, 9);
    const noPoolPrice = preMoney / shares;
    const laterNew = newMoney / noPoolPrice;
    const laterPool = (shares + laterNew) * p / (1 - p);
    const laterTotal = shares + laterNew + laterPool;
    expect(result.postMoneyPool.existingPercent).toBeCloseTo(shares / laterTotal * 100, 9);
    expect(result.postMoneyPool.investorPercent).toBeCloseTo(laterNew / laterTotal * 100, 9);
    expect(result.postMoneyPool.poolPercent).toBeCloseTo(laterPool / laterTotal * 100, 9);
    for (const allocation of [result.noPool, result.preMoneyPool, result.postMoneyPool]) {
      expect(allocation.existingPercent + allocation.investorPercent + allocation.poolPercent).toBeCloseTo(100, 9);
    }
    expect(result.postMoneyPool.existingPercent - result.preMoneyPool.existingPercent).toBeCloseTo(result.existingDifferencePoints, 9);
  }
});

test('a zero pool changes nothing and equal scaling of currency units changes no percentages', () => {
  const zero = good({ ...example, poolPercent: 0 });
  expect(zero.preMoneyPool).toEqual(zero.noPool);
  expect(zero.postMoneyPool).toEqual(zero.noPool);
  expect(zero.existingDifferencePoints).toBe(0);
  expect(good({ preMoney: 8_000_000, newMoney: 2_000_000, poolPercent: 10 })).toEqual(good());
  expect(example).toEqual({ preMoney: 8, newMoney: 2, poolPercent: 10 });
});

test('invalid, non-finite and impossible inputs fail instead of being clamped to a result', () => {
  for (const key of ['preMoney', 'newMoney', 'poolPercent'] as const) {
    for (const invalid of [NaN, Infinity, -Infinity, -1]) expect(compareOptionPoolDilution({ ...example, [key]: invalid }).ok).toBe(false);
  }
  for (const input of [
    { ...example, preMoney: 0 }, { ...example, newMoney: 0 },
    { ...example, poolPercent: 100 }, { ...example, poolPercent: 80 }, { ...example, poolPercent: 81 },
    { ...example, preMoney: Number.MAX_VALUE, newMoney: Number.MAX_VALUE },
    { ...example, preMoney: Number.MIN_VALUE, newMoney: 1 },
    { ...example, preMoney: 1, newMoney: Number.MIN_VALUE },
  ]) {
    const result = compareOptionPoolDilution(input);
    expect(result.ok).toBe(false);
    expect(result).not.toHaveProperty('noPool');
  }
});
